  ```python
  class Stack:
      """棧的基本實現"""
      def __init__(self):
          self._items: List[Any] = []
      
      def push(self, item: Any) -> None:
          """將元素添加到棧頂"""
          self._items.append(item)
      
      def pop(self) -> Optional[Any]:
          """移除並返回棧頂元素"""
          if self.is_empty():
              return None
          return self._items.pop()
      
      def peek(self) -> Optional[Any]:
          """查看棧頂元素但不移除"""
          if self.is_empty():
              return None
          return self._items[-1]
      
      def is_empty(self) -> bool:
          """檢查棧是否為空"""
          return len(self._items) == 0
      
      def size(self) -> int:
          """返回棧中元素個數"""
          return len(self._items)
  ```

- 🎯 **實現 Queue 類**

  ```python
  from collections import deque
  
  class Queue:
      """佇列實現（FIFO）"""
      
      def __init__(self):
          self._items: deque = deque()
      
      def enqueue(self, item: Any) -> None:
          """將元素添加到隊尾"""
          self._items.append(item)
      
      def dequeue(self) -> Optional[Any]:
          """移除並返回隊頭元素"""
          if self.is_empty():
              return None
          return self._items.popleft()
      
      def peek(self) -> Optional[Any]:
          """查看隊頭元素但不移除"""
          if self.is_empty():
              return None
          return self._items[0]
      
      def is_empty(self) -> bool:
          """檢查隊列是否為空"""
          return len(self._items) == 0
      
      def size(self) -> int:
          """返回隊列中元素個數"""
          return len(self._items)
      
      def get_all(self) -> List[Any]:
          """返回隊列中所有元素"""
          return list(self._items)
      
      def clear(self) -> None:
          """清空隊列"""
          self._items.clear()
  ```

  ```python
          task = self.dequeue()
          if task:
              task["status"] = "completed"
          return task

          """獲取所有待執行任務"""
  ```

- 📖 **BFS 算法與佇列**

  佇列最經典的應用就是廣度優先搜尋（BFS）：

  ```python
  class Graph:
      """簡單的圖實現"""
      
      def __init__(self):
          self.adjacency = {}  # {節點: [相鄰節點列表]}
      
      def add_edge(self, from_node, to_node):
          """添加邊"""
          if from_node not in self.adjacency:
              self.adjacency[from_node] = []
          self.adjacency[from_node].append(to_node)
      
      def bfs(self, start_node):
          """廣度優先搜尋"""
          visited = set()
          queue = Queue()
          queue.enqueue(start_node)
          visited.add(start_node)
          
          result = []
          while not queue.is_empty():
              node = queue.dequeue()
              result.append(node)
              
              # 訪問所有相鄰節點
              if node in self.adjacency:
                  for neighbor in self.adjacency[node]:
                      if neighbor not in visited:
                          queue.enqueue(neighbor)
                          visited.add(neighbor)
          
          return result
  
  # 測試
  graph = Graph()
  graph.add_edge("A", "B")
  graph.add_edge("A", "C")
  graph.add_edge("B", "D")
  graph.add_edge("C", "E")
  
  path = graph.bfs("A")
  print(f"BFS 順序: {path}")  # ['A', 'B', 'C', 'D', 'E']
  ```

- 🎯 **實現簡單 HashTable**

  ```python
  class HashTable:
      """簡單的雜湊表實現"""
      
      def __init__(self, capacity: int = 100):
          self._table: dict = {}
          self.capacity = capacity
          self._size = 0
      
      def _hash(self, key: str) -> int:
          """簡單的雜湊函數"""
          return hash(key) % self.capacity
      
      def put(self, key: str, value: Any) -> None:
          """將鍵值對存入雜湊表"""
          if key not in self._table:
              self._size += 1
          self._table[key] = value
      
      def get(self, key: str) -> Optional[Any]:
          """根據鍵獲取值"""
          return self._table.get(key)
      
      def remove(self, key: str) -> Optional[Any]:
          """移除指定鍵的元素"""
          if key in self._table:
              self._size -= 1
              return self._table.pop(key)
          return None
      
      def contains(self, key: str) -> bool:
          """檢查鍵是否存在"""
          return key in self._table
      
      def size(self) -> int:
          """返回雜湊表中元素個數"""
          return self._size
      
      def keys(self) -> List[str]:
          """返回所有鍵"""
          return list(self._table.keys())
      
      def values(self) -> List[Any]:
          """返回所有值"""
          return list(self._table.values())
      
      def items(self) -> List[tuple]:
          """返回所有鍵值對"""
  ```

  ```python
  class ConversationAgent:
      """對話 Agent 類"""
      
      def __init__(self, api_key: Optional[str] = None):
          self.client = Anthropic(api_key=api_key)
          self.conversation_stack = ConversationStack()
          self.model = "claude-3-5-sonnet-20241022"
      
      def add_user_message(self, content: str) -> None:
          """添加用戶消息"""
          self.conversation_stack.add_message("user", content)
      
      def add_assistant_message(self, content: str) -> None:
          """添加 Assistant 消息"""
          self.conversation_stack.add_message("assistant", content)
      
      def get_conversation_history(self) -> List[Dict[str, str]]:
          """獲取完整對話歷史"""
          return self.conversation_stack.get_conversation_history()
      
      def generate_response(self, user_input: str) -> str:
          """
          生成回應
          
          步驟：
          1. 添加用戶輸入到棧
          2. 調用 Claude API
          3. 添加回應到棧
          4. 返回回應文本
          """
          self.add_user_message(user_input)
          
          messages = self.get_conversation_history()
          
          response = self.client.messages.create(
              model=self.model,
              max_tokens=1024,
              system="你是一個有幫助的 AI 助手。用繁體中文回應。",
              messages=messages
          )
          
          assistant_response = response.content[0].text
          self.add_assistant_message(assistant_response)
          
          return assistant_response
  ```

  ```python
  class MultiSessionAgent:
      """多會話 Agent 管理器"""
      
      def __init__(self):
          self.agents: Dict[str, ConversationAgent] = {}
      
      def create_session(self, session_id: str) -> ConversationAgent:
          """為新用戶建立會話"""
          if session_id not in self.agents:
              self.agents[session_id] = ConversationAgent()
          return self.agents[session_id]
      
      def get_session(self, session_id: str) -> Optional[ConversationAgent]:
          """獲取現有會話"""
          return self.agents.get(session_id)
      
      def session_exists(self, session_id: str) -> bool:
          """檢查會話是否存在"""
          return session_id in self.agents
      
      def delete_session(self, session_id: str) -> bool:
          """刪除會話"""
          if session_id in self.agents:
              del self.agents[session_id]
              return True
          return False
      
      def generate_response(self, session_id: str, user_input: str) -> str:
          """為特定會話生成回應"""
          if not self.session_exists(session_id):
              self.create_session(session_id)
          
          agent = self.get_session(session_id)
          return agent.generate_response(user_input)
      
      def get_conversation_history(self, session_id: str) -> List[Dict[str, str]]:
          """獲取特定會話的對話歷史"""
          agent = self.get_session(session_id)
          if agent:
              return agent.get_conversation_history()
          return []
      
      def get_all_sessions(self) -> List[str]:
          """獲取所有活躍會話 ID"""
          return list(self.agents.keys())
  ```

## 常見問題與解決方案

### Q: 我的代碼運行不了，怎麼辦？

**A:** 按以下步驟診斷：
1. 檢查是否有語法錯誤：`python -m py_compile filename.py`
2. 查看錯誤信息：錯誤信息會告訴你問題在哪一行
3. 使用 debugger：在代碼中加 `import pdb; pdb.set_trace()`
4. 查看測試輸出：運行單元測試看具體失敗的地方

### Q: 如何測試我的代碼？

**A:** 三種方式：
1. **單元測試**：在本文檔中提供了每個方法的測試代碼
2. **集成測試**：測試多個模塊一起工作
3. **手動測試**：用 Swagger UI 或 curl 測試 API

### Q: 我想添加新功能，怎麼設計？

**A:** 遵循以下步驟：
1. 先設計數據結構
2. 寫好測試（測試驅動開發）
3. 實現功能
4. 測試通過
5. 集成到系統中
