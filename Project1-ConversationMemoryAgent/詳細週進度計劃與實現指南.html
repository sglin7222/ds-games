class Stack:
    """棧的基本實現"""
    def __init__(self):
        self._items: List[Any] = []

    def push(self, item: Any) -> None:
        """將元素添加到棧頂"""
        self._items.append(item)

    def pop(self) -> Optional[Any]:
        """移除並返回棧頂元素"""
        if self.is_empty():
            return None
        return self._items.pop()

    def peek(self) -> Optional[Any]:
        """查看棧頂元素但不移除"""
        if self.is_empty():
            return None
        return self._items[-1]

    def is_empty(self) -> bool:
        """檢查棧是否為空"""
        return len(self._items) == 0

    def size(self) -> int:
        """返回棧中元素個數"""
        return len(self._items)
  • Stack 類有 5 個基本方法
  • 可以運行 stack.push(1) 不報錯

class EditHistory:
    """編輯歷史，支持撤回"""
    def __init__(self):
        self.history = Stack()

    def type_text(self, text: str) -> None:
        """鍵入文本"""
        self.history.push(text)

    def undo(self) -> Optional[str]:
        """撤回"""
        return self.history.pop()

    def get_current(self) -> str:
        """獲取當前文本"""
        if self.history.is_empty():
            return ""
        return " ".join(self.history.get_all())

📖 複雜度分析： 理解為什麼棧操作是 O(1)

push(item):     O(1)    - 只需 append，一次操作
pop():          O(1)    - 只需移除最後元素，一次操作
peek():         O(1)    - 只需訪問最後元素，一次操作
is_empty():     O(1)    - 只需檢查長度，一次操作
size():         O(1)    - 直接返回長度，一次操作
get_all():      O(n)    - 需要複製 n 個元素

class ConversationStack(Stack):
    """對話棧的特殊實現"""

    def add_message(self, role: str, content: str) -> None:
        """添加對話消息"""
        message = {
            "role": role,      # "user" 或 "assistant"
            "content": content
        }
        self.push(message)

    def get_conversation_history(self) -> List[dict]:
        """獲取完整對話歷史"""
        return self.get_all()

    def undo_last_message(self) -> Optional[dict]:
        """撤回最後一條消息"""
        return self.pop()

    def get_recent_messages(self, n: int = 5) -> List[dict]:
        """獲取最近 n 條消息"""
        all_messages = self.get_all()
        return all_messages[-n:] if len(all_messages) > 0 else []

import time
from collections import deque

# 測試 List
def test_list_performance():
    lst = []
    start = time.time()
    for i in range(100000):
        lst.append(i)
    for i in range(100000):
        lst.pop(0)  # ❌ 每次都要移動所有元素
    return time.time() - start

# 測試 Deque
def test_deque_performance():
    dq = deque()
    start = time.time()
    for i in range(100000):
        dq.append(i)
    for i in range(100000):
        dq.popleft()  # ✅ O(1) 操作
    return time.time() - start

list_time = test_list_performance()
deque_time = test_deque_performance()

print(f"List 耗時: {list_time:.4f}秒")
print(f"Deque 耗時: {deque_time:.4f}秒")

from collections import deque

class Queue:
    """佇列實現（FIFO）"""

    def __init__(self):
        self._items: deque = deque()

    def enqueue(self, item: Any) -> None:
        """將元素添加到隊尾"""
        self._items.append(item)

    def dequeue(self) -> Optional[Any]:
        """移除並返回隊頭元素"""
        if self.is_empty():
            return None
        return self._items.popleft()

    def peek(self) -> Optional[Any]:
        """查看隊頭元素但不移除"""
        if self.is_empty():
            return None
        return self._items[0]

    def is_empty(self) -> bool:
        """檢查隊列是否為空"""
        return len(self._items) == 0

    def size(self) -> int:
        """返回隊列中元素個數"""
        return len(self._items)

    def get_all(self) -> List[Any]:
        """返回隊列中所有元素"""
        return list(self._items)

    def clear(self) -> None:
        """清空隊列"""
        self._items.clear()

class TaskQueue(Queue):
    """任務隊列的特殊實現"""

    def add_task(self, task_id: str, task_type: str, data: dict) -> None:
        """添加任務到隊列"""
        task = {
            "id": task_id,
            "type": task_type,
            "data": data,
            "status": "pending"
        }
        self.enqueue(task)

    def get_next_task(self) -> Optional[dict]:
        """獲取下一個待執行的任務"""
        return self.peek()

    def complete_task(self) -> Optional[dict]:
        """完成並移除下一個任務"""
        task = self.dequeue()
        if task:
            task["status"] = "completed"
        return task

    def get_pending_tasks(self) -> List[dict]:
        """獲取所有待執行任務"""
        return self.get_all()

📅 第 3 天 - BFS 算法

class Graph:
    """簡單的圖實現"""

    def __init__(self):
        self.adjacency = {}  # {節點: [相鄰節點列表]}

    def add_edge(self, from_node, to_node):
        """添加邊"""
        if from_node not in self.adjacency:
            self.adjacency[from_node] = []
        self.adjacency[from_node].append(to_node)

    def bfs(self, start_node):
        """廣度優先搜尋"""
        visited = set()
        queue = Queue()
        queue.enqueue(start_node)
        visited.add(start_node)

        result = []
        while not queue.is_empty():
            node = queue.dequeue()
            result.append(node)

            # 訪問所有相鄰節點
            if node in self.adjacency:
                for neighbor in self.adjacency[node]:
                    if neighbor not in visited:
                        queue.enqueue(neighbor)
                        visited.add(neighbor)

        return result

# 測試

class HashTable:
    def __init__(self, capacity: int = 100):
        self._table: dict = {}
        self.capacity = capacity
        self._size = 0

    def put(self, key: str, value: Any) -> None:
        if key not in self._table:
            self._size += 1
        self._table[key] = value

    def get(self, key: str) -> Optional[Any]:
        return self._table.get(key)

    def remove(self, key: str) -> Optional[Any]:
        if key in self._table:
            self._size -= 1
            return self._table.pop(key)
        return None

class SessionCache(HashTable):
    def create_session(self, user_id: str, session_data: dict) -> None:
